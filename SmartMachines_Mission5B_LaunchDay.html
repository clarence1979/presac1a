# Mission 5B - RETURN FIRE: Camera-Verified Intruder Alarm
# Raspberry Pi 5 + Kali Linux + GrovePi+ + Logitech USB webcam + USB foam launcher
#
# Run with:   sudo python3 mission_5b_camera.py        (Ctrl+C to stop)
#
# Two ways to run it:
#   A  Sensor-only alarm (no API key, no internet needed):   use_camera_check = False
#   B  Camera-verified alarm (needs a teacher-issued OpenAI key, the webcam and the internet):
#        use_camera_check = True   and   sudo apt install python3-opencv python3-requests
#
# Plug the parts into the GrovePi+ like this:
#   Green LED -> D2    Red LED -> D3    Ultrasonic ranger -> D5    Speaker -> D6
#   Buzzer -> D7       Temp & humidity (blue DHT11) -> D8          Light sensor -> A0
#   LCD RGB Backlight -> any I2C port
#   Logitech webcam and the foam launcher -> the Pi's USB ports (NOT the GrovePi+)
#
# SAFETY: foam darts only. The launcher stays OFF (cannon_enabled = False) unless your
# teacher says otherwise, and then it fires at an empty target - never at a face.

import os
import fcntl
import struct
import time
import signal
import getpass

# ---------- Settings you can change ----------
use_camera_check = False          # True = ask the AI to confirm a person before the alarm (needs a key)
openai_api_key = "PASTE_YOUR_OPENAI_API_KEY_HERE"
openai_model = "gpt-4.1-mini"
camera_number = 0                 # try 1 if another camera is connected

secret_pin = "1234"               # the PIN that stops the alarm
arming_delay = 10                 # seconds you get to leave the room
distance_change = 20              # someone this many cm closer than normal = intruder
light_change = 150                # light level changing by this much = intruder
heat_limit = 40                   # temperature (C) that counts as a possible fire
sound_on = True                   # set to False to test quietly in class
log_file_name = "intrusion_log.txt"

cannon_enabled = False            # TEACHER ONLY: True fires foam darts at a safe, empty target
rockets_loaded = 4                # how many darts you put in the launcher
rockets_per_alarm = 1             # how many to fire each time the alarm goes off
aim_right = 3.0                   # seconds to turn right from the parked position to aim
aim_up = 0.8                      # seconds to tilt up from the parked position

# ---------- Ports ----------
green_led_port = 2
red_led_port = 3
ranger_port = 5
speaker_port = 6                  # must be D3, D5 or D6 - only these ports can play a tone
buzzer_port = 7
dht_port = 8
light_port = 0                    # A0

# ---------- Launcher command numbers ----------
cannon_down = 1
cannon_up = 2
cannon_left = 4
cannon_right = 8
cannon_fire = 16
cannon_stop = 32


# ---------- Connecting to the devices ----------
# Every device on the I2C wires has its own address (a number).
def open_device(address):
    device = os.open("/dev/i2c-1", os.O_RDWR)
    fcntl.ioctl(device, 0x0703, address)     # 0x0703 means "talk to this address"
    return device


# ---------- GrovePi+ helpers ----------
# Every GrovePi+ message is 5 bytes: 1, the command number, the port, a value, 0
def grovepi_send(command, port, value):
    message = struct.pack("BBBBB", 1, command, port, value, 0)
    os.write(grovepi, message)
    time.sleep(0.02)


# The GrovePi+ answers with 3 bytes: a label, then a number split into two bytes
def grovepi_answer():
    reply = os.read(grovepi, 3)
    label, high, low = struct.unpack("BBB", reply)
    return high * 256 + low


def make_output(port):
    grovepi_send(5, port, 1)         # command 5 = set a port to be an output


def turn_on(port):
    grovepi_send(2, port, 1)         # command 2 = digital write, 1 = on


def turn_off(port):
    grovepi_send(2, port, 0)         # 0 = off


def read_distance():
    grovepi_send(7, ranger_port, 0)  # command 7 = ultrasonic read
    time.sleep(0.1)                  # wait for the echo
    return grovepi_answer()


def read_light():
    grovepi_send(3, light_port, 0)   # command 3 = analog read (0 = dark, 1023 = bright)
    time.sleep(0.05)
    return grovepi_answer()


def read_temperature_and_humidity():
    grovepi_send(40, dht_port, 0)    # command 40 = temp & humidity, 0 = blue DHT11
    time.sleep(0.6)                  # this sensor is slow
    reply = os.read(grovepi, 9)
    # The reply is 9 bytes: a label, then two decimal numbers of 4 bytes each.
    label, temperature, humidity = struct.unpack("<Bff", reply)
    return temperature, humidity


# ---------- Speaker helpers ----------
def speaker_on():
    grovepi_send(4, speaker_port, 128)   # command 4 = analog write: makes a tone


def speaker_off():
    grovepi_send(4, speaker_port, 0)


def beep(seconds):
    if sound_on:
        speaker_on()
        time.sleep(seconds)
        speaker_off()


def siren(times):
    # "nee-naw": the speaker (low note) and the buzzer (high note) take turns
    if sound_on:
        for count in range(times):
            turn_off(buzzer_port)
            speaker_on()
            time.sleep(0.3)
            speaker_off()
            turn_on(buzzer_port)
            time.sleep(0.3)


# ---------- LCD helpers ----------
# The LCD has two addresses: one for the text, one for the coloured backlight.
def lcd_command(code):
    os.write(lcd_text, struct.pack("BB", 0x80, code))
    time.sleep(0.005)


def lcd_setup():
    lcd_command(0x01)       # clear the screen
    time.sleep(0.05)
    lcd_command(0x0C)       # screen on, no cursor
    lcd_command(0x28)       # use both lines
    os.write(lcd_backlight, struct.pack("BB", 0, 0))      # wake up the backlight chip
    os.write(lcd_backlight, struct.pack("BB", 1, 0))
    os.write(lcd_backlight, struct.pack("BB", 8, 0xAA))


def lcd_colour(red, green, blue):
    # each colour is a number from 0 (off) to 255 (full)
    os.write(lcd_backlight, struct.pack("BB", 4, red))
    os.write(lcd_backlight, struct.pack("BB", 3, green))
    os.write(lcd_backlight, struct.pack("BB", 2, blue))


def lcd_write_line(code, text):
    lcd_command(code)                    # 0x80 = start of line 1, 0xC0 = start of line 2
    text = text.ljust(16)                # pad with spaces to clear old letters
    for letter in text:
        # ord() turns a letter into its number code, which is what the LCD needs
        os.write(lcd_text, struct.pack("BB", 0x40, ord(letter)))


def lcd_show(line1, line2):
    # Each line holds 16 characters - keep messages short!
    lcd_write_line(0x80, line1)
    lcd_write_line(0xC0, line2)


# ---------- Launcher helpers ----------
# The launcher is a USB device. Every message is 8 bytes: 2, command, 0, 0, 0, 0, 0, 0
def cannon_send(command):
    message = struct.pack("BBBBBBBB", 2, command, 0, 0, 0, 0, 0, 0)
    cannon.ctrl_transfer(0x21, 0x09, 0, 0, message)


def cannon_move(command, seconds):
    cannon_send(command)             # start moving...
    time.sleep(seconds)              # ...for this long...
    cannon_send(cannon_stop)         # ...then stop


def cannon_light(on):
    # 3 means "the LED": on = 1 lights it up, on = 0 switches it off
    message = struct.pack("BBBBBBBB", 3, on, 0, 0, 0, 0, 0, 0)
    cannon.ctrl_transfer(0x21, 0x09, 0, 0, message)


def cannon_shoot(rockets):
    for count in range(rockets):
        cannon_send(cannon_fire)
        time.sleep(4.5)              # it takes about 4 seconds to fire and reload


def cannon_park():
    # swing all the way down and left, so every run starts from the same place
    cannon_move(cannon_down, 2)
    cannon_move(cannon_left, 8)


def cannon_aim():
    # from the parked position, turn and tilt to point at the safe target
    cannon_move(cannon_right, aim_right)
    cannon_move(cannon_up, aim_up)


# ---------- Camera + AI check ----------
def person_in_camera():
    # These two libraries are only needed for the camera check, so they are
    # loaded here. Install them with: sudo apt install python3-opencv python3-requests
    import base64
    import cv2
    import requests

    if openai_api_key == "PASTE_YOUR_OPENAI_API_KEY_HERE":
        print("No API key in the settings. Alarm stays off.")
        return False

    # Step 1: take ONE photo, only after the sensors noticed something.
    camera = cv2.VideoCapture(camera_number, cv2.CAP_V4L2)
    if not camera.isOpened():
        print("Camera not available. Alarm stays off.")
        return False
    time.sleep(0.4)
    picture_ok, picture = camera.read()
    camera.release()
    if not picture_ok:
        print("Could not take a photo. Alarm stays off.")
        return False

    # Step 2: make the photo small to save API credits.
    picture = cv2.resize(picture, (640, 480))
    image_ok, image_bytes = cv2.imencode(".jpg", picture)
    if not image_ok:
        return False
    photo_text = base64.b64encode(image_bytes.tobytes()).decode("utf-8")
    photo_url = "data:image/jpeg;base64," + photo_text

    # Step 3: ask the AI a YES / NO question about the photo.
    # It checks whether a person is visible. It does not work out who they are.
    headers = {"Authorization": "Bearer " + openai_api_key,
               "Content-Type": "application/json"}
    question = ("Is a real human person visible in this camera photo? Reply with only YES or NO. "
                "Do not count a person shown only on a poster, photo, or screen. If uncertain reply NO.")
    data = {"model": openai_model,
            "max_output_tokens": 20,
            "input": [{"role": "user", "content": [
                {"type": "input_text", "text": question},
                {"type": "input_image", "image_url": photo_url, "detail": "low"}]}]}

    # If the internet is down this line stops the program with a readable error.
    response = requests.post("https://api.openai.com/v1/responses",
                             headers=headers, json=data, timeout=30)
    if response.status_code != 200:
        print("AI error:", response.status_code, response.text[:180])
        return False

    answer = ""
    result = response.json()
    for item in result.get("output", []):
        if item.get("type") == "message":
            for part in item.get("content", []):
                if part.get("type") == "output_text":
                    answer = part.get("text", "").strip().upper()
    print("Camera result:", answer)
    return answer == "YES"


# ---------- Log file ----------
def write_log(message):
    log_file = open(log_file_name, "a")      # "a" = add to the end of the file
    log_file.write(time.strftime("%Y-%m-%d %H:%M:%S") + "  " + message + "\n")
    log_file.close()


# ---------- Set up ----------
signal.signal(signal.SIGINT, signal.SIG_DFL)     # make Ctrl+C stop the program quietly

print("Connecting to the GrovePi+ and the LCD...")
grovepi = open_device(0x04)
lcd_text = open_device(0x3E)
lcd_backlight = open_device(0x62)

cannon = None
if cannon_enabled:
    import usb.core                      # only needed for the launcher: sudo apt install python3-usb
    cannon = usb.core.find(idVendor=0x2123, idProduct=0x1010)
    if cannon is None:
        print("Launcher not found. Is it plugged in? Try: lsusb   (alarm still works)")
        cannon_enabled = False
    else:
        if cannon.is_kernel_driver_active(0):
            cannon.detach_kernel_driver(0)   # Linux grabs it as if it were a keyboard - let go
        cannon.set_configuration()

make_output(green_led_port)
make_output(red_led_port)
make_output(buzzer_port)
make_output(speaker_port)
turn_off(green_led_port)
turn_off(red_led_port)
turn_off(buzzer_port)
speaker_off()
lcd_setup()

rockets_left = rockets_loaded
temperature = 0
humidity = 0

if cannon_enabled:
    print("Parking and aiming the launcher at the safe target (about 15 seconds)...")
    cannon_park()
    cannon_aim()

if use_camera_check:
    print("Camera check is ON: the alarm only sounds when the AI confirms a person.")
else:
    print("Camera check is OFF: the alarm sounds after two unusual sensor readings.")
print("Ready. Press Ctrl+C to stop.")

# ---------- Main program ----------
while True:

    # 1. ARMING - countdown so you can leave the room
    lcd_colour(255, 120, 0)                          # orange
    seconds_left = arming_delay
    while seconds_left > 0:
        lcd_show("Arming...", "Leave in " + str(seconds_left) + " sec")
        turn_on(green_led_port)                      # blink the green LED...
        beep(0.1)                                    # ...and beep
        time.sleep(0.4)
        turn_off(green_led_port)
        time.sleep(0.5)
        seconds_left = seconds_left - 1

    # 2. LEARNING - what does the room look like when nobody is there?
    normal_distance = read_distance()
    if normal_distance == 0:                         # 0 means nothing in range
        normal_distance = 400
    normal_light = read_light()
    write_log("ARMED  (normal distance " + str(normal_distance) + " cm, light " + str(normal_light) + ")")
    print("Armed. Normal distance:", normal_distance, "cm   Normal light:", normal_light)
    lcd_colour(0, 90, 255)                           # blue
    turn_on(green_led_port)
    beep(0.5)                                        # one long beep = armed

    # 3. ARMED - keep checking until something changes (and, if on, the camera agrees)
    strikes = 0
    loops = 0
    person_found = False
    reason = ""
    while not person_found:

        # The temp & humidity sensor is slow, so only check it every 25 loops
        if loops % 25 == 0:
            new_temperature, new_humidity = read_temperature_and_humidity()
            if new_temperature > 0 and new_temperature < 60:  # ignore a failed reading
                temperature = round(new_temperature)
                humidity = round(new_humidity)
        loops = loops + 1

        distance = read_distance()
        light = read_light()
        lcd_show("ARMED " + str(temperature) + "C " + str(humidity) + "%",
                 str(distance) + "cm light " + str(light))

        if distance > 0 and distance < normal_distance - distance_change:
            strikes = strikes + 1
            reason = "Movement " + str(distance) + "cm"
        elif abs(light - normal_light) > light_change:
            strikes = strikes + 1
            reason = "Light changed"
        elif temperature > heat_limit:
            strikes = strikes + 1
            reason = "FIRE? " + str(temperature) + "C"
        else:
            strikes = 0                              # back to normal: start counting again

        if strikes >= 2:                             # 2 in a row stops false alarms
            if use_camera_check:
                print("Sensor change! Checking the camera...")
                lcd_colour(255, 180, 0)              # amber
                lcd_show("CHECKING CAMERA", reason)
                person_found = person_in_camera()
                strikes = 0
                if not person_found:
                    print("No person confirmed. The alarm stays off.")
                    write_log("Sensor change (" + reason + "), person NOT confirmed")
                    lcd_colour(0, 90, 255)
                    time.sleep(5)                    # don't photograph again straight away
            else:
                person_found = True                  # sensor-only mode: two strikes = alarm

        time.sleep(0.2)

    # 4. ALARM!
    turn_off(green_led_port)
    turn_on(red_led_port)
    lcd_colour(255, 0, 0)                            # red
    lcd_show("!!! ALARM !!!", reason)
    if use_camera_check:
        write_log("ALARM - person confirmed by camera (" + reason + ")")
    else:
        write_log("ALARM - " + reason)
    print()
    print("!!! ALARM !!!", reason)
    siren(2)                                         # nee-naw twice...
    if cannon_enabled:
        cannon_light(1)                              # ...light up the launcher...
        if rockets_left >= rockets_per_alarm:
            print("FIRE!")
            cannon_shoot(rockets_per_alarm)          # ...and FIRE at the safe target
            rockets_left = rockets_left - rockets_per_alarm
            write_log("FIRED " + str(rockets_per_alarm) + " rocket(s), " + str(rockets_left) + " left")
        else:
            print("Out of rockets!")
    siren(3)                                         # nee-naw 3 more times...
    if sound_on:
        turn_on(buzzer_port)                         # ...then the buzzer stays on

    # 5. DISARM - keep asking until the right PIN is typed
    wrong_tries = 0
    typed_pin = getpass.getpass("Enter PIN to stop the alarm: ")
    while typed_pin != secret_pin:
        wrong_tries = wrong_tries + 1
        write_log("WRONG PIN entered (attempt " + str(wrong_tries) + ")")
        print("Wrong PIN!")
        lcd_show("WRONG PIN!", "Attempts: " + str(wrong_tries))
        siren(3)
        if sound_on:
            turn_on(buzzer_port)
        typed_pin = getpass.getpass("Enter PIN to stop the alarm: ")

    turn_off(buzzer_port)
    turn_off(red_led_port)
    if cannon_enabled:
        cannon_light(0)
    lcd_colour(0, 255, 0)                            # green
    lcd_show("Disarmed", "Wrong PINs: " + str(wrong_tries))
    write_log("DISARMED after " + str(wrong_tries) + " wrong PIN(s)")
    print("Alarm stopped.")
    beep(0.1)                                        # two short "all clear" beeps
    time.sleep(0.1)
    beep(0.1)
    if cannon_enabled and rockets_left < rockets_per_alarm:
        input("Reload the launcher, then press Enter to arm again...")
        rockets_left = rockets_loaded
    else:
        input("Press Enter to arm the alarm again...")
