# Launch Day - SENSOR CHECK
# GrovePi+ on a Raspberry Pi 5 (Kali Linux)
#
# What it does: prints what your sensors can see, once a second, so your team
# can find two numbers: what "normal" looks like, and what "something is
# happening" looks like. Your threshold goes in between.
#
# Plug in only the sensors your invention needs:
#   Ultrasonic ranger    ->  D5
#   Light sensor         ->  A0
#   Temp & humidity      ->  D8   (the blue DHT11 sensor)
# Then set the matching use_ setting below to True, and the others to False.
#
# Run with:   sudo python3 sensor_check.py        (Ctrl+C to stop)

import os
import fcntl
import struct
import time
import signal

# ---------- Settings you can change ----------
use_ranger = True         # True if the ultrasonic ranger is plugged into D5
use_light = True          # True if the light sensor is plugged into A0
use_temp = False          # True if the blue temp & humidity sensor is plugged into D8

ranger_port = 5
light_port = 0            # A0
dht_port = 8
wait_time = 1             # seconds between readings


# ---------- Connecting to the GrovePi+ ----------
# The GrovePi+ listens on the I2C wires at address 4.
def open_device(address):
    device = os.open("/dev/i2c-1", os.O_RDWR)
    fcntl.ioctl(device, 0x0703, address)     # 0x0703 means "talk to this address"
    return device


# ---------- GrovePi+ helpers ----------
# Every GrovePi+ message is 5 bytes: 1, the command number, the port, a value, 0
def grovepi_send(command, port, value):
    message = struct.pack("BBBBB", 1, command, port, value, 0)
    os.write(grovepi, message)
    time.sleep(0.02)


# The GrovePi+ answers with 3 bytes: a label, then a number split into two bytes
def grovepi_answer():
    reply = os.read(grovepi, 3)
    label, high, low = struct.unpack("BBB", reply)
    return high * 256 + low


def read_distance():
    grovepi_send(7, ranger_port, 0)  # command 7 = ultrasonic read
    time.sleep(0.1)                  # wait for the echo to come back
    return grovepi_answer()


def read_light():
    grovepi_send(3, light_port, 0)   # command 3 = analog read (0 = dark, 1023 = bright)
    time.sleep(0.05)
    return grovepi_answer()


def read_temperature_and_humidity():
    grovepi_send(40, dht_port, 0)    # command 40 = temp & humidity, 0 = blue DHT11
    time.sleep(0.6)                  # this sensor is slow
    reply = os.read(grovepi, 9)
    # The reply is 9 bytes: a label, then two decimal numbers of 4 bytes each.
    label, temperature, humidity = struct.unpack("<Bff", reply)
    return temperature, humidity


# ---------- Set up ----------
signal.signal(signal.SIGINT, signal.SIG_DFL)     # make Ctrl+C stop the program quietly

grovepi = open_device(0x04)

# Lowest and highest readings so far. They start at "impossible" numbers,
# so the very first real reading replaces them.
lowest_distance = 9999
highest_distance = 0
lowest_light = 9999
highest_light = 0
lowest_temp = 99
highest_temp = 0

reading = 0

print("Sensor check started. Press Ctrl+C to stop.")
print("Step 1: leave everything NORMAL for 5 readings. Write the numbers down.")
print("Step 2: make your thing HAPPEN (walk past, shine a torch, warm hand near the sensor).")
print()

# ---------- Main program ----------
while True:
    reading = reading + 1
    line = "Reading " + str(reading) + ":"

    if use_ranger:
        distance = read_distance()
        line = line + "   distance " + str(distance) + " cm"
        if distance > 0:                         # 0 means nothing in range, so skip it
            if distance < lowest_distance:
                lowest_distance = distance
            if distance > highest_distance:
                highest_distance = distance

    if use_light:
        light = read_light()
        line = line + "   light " + str(light)
        if light < lowest_light:
            lowest_light = light
        if light > highest_light:
            highest_light = light

    if use_temp:
        temperature, humidity = read_temperature_and_humidity()
        if temperature > 0 and temperature < 60:  # ignore a failed reading
            temperature = round(temperature)
            humidity = round(humidity)
            line = line + "   temp " + str(temperature) + " C   humidity " + str(humidity) + " %"
            if temperature < lowest_temp:
                lowest_temp = temperature
            if temperature > highest_temp:
                highest_temp = temperature
        else:
            line = line + "   temp (missed a reading - that's normal for this sensor)"

    print(line)

    # Every 5 readings, show the lowest and highest numbers so far
    if reading % 5 == 0:
        summary = "    lowest to highest so far:"
        if use_ranger and highest_distance > 0:
            summary = summary + "   distance " + str(lowest_distance) + " to " + str(highest_distance) + " cm"
        if use_light:
            summary = summary + "   light " + str(lowest_light) + " to " + str(highest_light)
        if use_temp and highest_temp > 0:
            summary = summary + "   temp " + str(lowest_temp) + " to " + str(highest_temp) + " C"
        print(summary)
        print()

    time.sleep(wait_time)
